/* =========================================================
   BASIC HELPERS
   ========================================================= */

function cleanString(value, max = 5000) {
    if (typeof value !== 'string') {
        throw new Error('Expected text');
    }

    const v = value.trim();

    if (!v || v.length > max) {
        throw new Error('Invalid text length');
    }

    return v;
}

function validStatus(value) {
    return [
        'locked',
        'provisional',
        'idea',
        'rejected'
    ].includes(value);
}

function escapeHtml(value) {
    return String(value)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
}

function escapeRegExp(value) {
    return String(value).replace(
        /[.*+?^${}()|[\]\\]/g,
        '\\$&'
    );
}

function slugify(value) {
    return String(value)
        .toLowerCase()
        .trim()
        .replace(/&/g, ' and ')
        .replace(/[^a-z0-9]+/g, '-')
        .replace(/^-+|-+$/g, '');
}

function timestamp() {
    const d = new Date();

    const pad = n =>
        String(n).padStart(2, '0');

    return [
        d.getFullYear(),
        pad(d.getMonth() + 1),
        pad(d.getDate())
    ].join('') +
    '-' +
    [
        pad(d.getHours()),
        pad(d.getMinutes()),
        pad(d.getSeconds())
    ].join('');
}

function generateId() {
    return (
        'canon-' +
        Date.now().toString(36) +
        '-' +
        Math.random()
            .toString(36)
            .slice(2, 8)
    );
}


/* =========================================================
   CANON FILE
   ========================================================= */

function ensureCanonFile() {
    if (!fs.existsSync(canonFile)) {
        throw new Error(
            'Master Canon file not found at ' +
            canonFile
        );
    }
}

function readCanon() {
    ensureCanonFile();

    return fs.readFileSync(
        canonFile,
        'utf8'
    );
}

function writeCanon(html) {
    fs.writeFileSync(
        canonFile,
        html,
        'utf8'
    );
}


/* =========================================================
   BACKUPS
   ========================================================= */

function backupCanon() {
    ensureCanonFile();

    const backupName =
        `master-canon-${timestamp()}.html`;

    const destination =
        path.join(
            backupDir,
            backupName
        );

    fs.copyFileSync(
        canonFile,
        destination
    );

    return backupName;
}


/* =========================================================
   REVISION HISTORY
   ========================================================= */

function readRevisions() {
    if (!fs.existsSync(revisionFile)) {
        return [];
    }

    try {
        const data = JSON.parse(
            fs.readFileSync(
                revisionFile,
                'utf8'
            )
        );

        return Array.isArray(data)
            ? data
            : [];
    } catch {
        return [];
    }
}

function saveRevisions(rows) {
    fs.writeFileSync(
        revisionFile,
        JSON.stringify(
            rows.slice(0, 500),
            null,
            2
        ),
        'utf8'
    );
}

function addRevision({
    action,
    target,
    targetId,
    details,
    username
}) {
    const rows = readRevisions();

    rows.unshift({
        action,
        target_type: target || 'canon',
        target_id: targetId || null,
        details: details || '',
        username: username || 'owner',
        created_at: new Date().toISOString()
    });

    saveRevisions(rows);
}


/* =========================================================
   SAFE WRITE
   ========================================================= */

function modifyCanon(callback) {
    const current = readCanon();

    const backup = backupCanon();

    try {
        const updated = callback(current);

        if (
            typeof updated !== 'string' ||
            updated === current
        ) {
            throw new Error(
                'No changes were made to the Master Canon.'
            );
        }

        writeCanon(updated);

        return {
            html: updated,
            backup
        };

    } catch (error) {
        /*
         * If modification fails, restore the original
         * automatically from the backup.
         */
        fs.copyFileSync(
            path.join(backupDir, backup),
            canonFile
        );

        throw error;
    }
}


/* =========================================================
   SECTION PARSING
   ========================================================= */

function getSections() {
    const html = readCanon();

    const sections = [];

    const regex =
        /<div\s+id="([^"]+)"\s+class="section(?:\s+[^"]*)?"[^>]*>([\s\S]*?)(?=<div\s+id="[^"]+"\s+class="section(?:\s|")|<\/div>\s*<!-- =====================================================\s+FOOTER)/gi;

    let match;

    while ((match = regex.exec(html)) !== null) {
        const slug = match[1];
        const body = match[2];

        const titleMatch =
            body.match(
                /<div\s+class="section-title">\s*([\s\S]*?)\s*<\/div>/i
            );

        const title = titleMatch
            ? stripHtml(titleMatch[1]).trim()
            : slug;

        sections.push({
            slug,
            name: title
        });
    }

    /*
     * Fallback parser.
     * This handles the current document's structure
     * even if the large regex above cannot determine
     * the exact section boundary.
     */
    if (!sections.length) {
        const fallback =
            /<div\s+id="([^"]+)"\s+class="section[^"]*"[\s\S]*?<div\s+class="section-title">\s*([\s\S]*?)\s*<\/div>/gi;

        while ((match = fallback.exec(html)) !== null) {
            sections.push({
                slug: match[1],
                name: stripHtml(match[2]).trim()
            });
        }
    }

    return uniqueSections(sections);
}

function uniqueSections(sections) {
    const seen = new Set();

    return sections.filter(section => {
        if (seen.has(section.slug)) {
            return false;
        }

        seen.add(section.slug);
        return true;
    });
}

function stripHtml(value) {
    return String(value)
        .replace(/<[^>]*>/g, '')
        .replace(/\s+/g, ' ')
        .trim();
}

function findSectionByName(name) {
    const q = cleanString(name, 200)
        .toLowerCase()
        .trim();

    const sections = getSections();

    return (
        sections.find(
            s => s.slug.toLowerCase() === q
        ) ||
        sections.find(
            s => s.name.toLowerCase() === q
        ) ||
        sections.find(
            s => s.name.toLowerCase().includes(q)
        )
    );
}


/* =========================================================
   SECTION HTML
   ========================================================= */

function findSectionRange(html, sectionSlug) {
    const startRegex = new RegExp(
        `<div\\s+id="${escapeRegExp(sectionSlug)}"\\s+class="section[^"]*"[^>]*>`,
        'i'
    );

    const startMatch =
        startRegex.exec(html);

    if (!startMatch) {
        throw new Error(
            `Section "${sectionSlug}" not found in Master Canon.`
        );
    }

    const start = startMatch.index;

    /*
     * We locate the next top-level section by searching
     * for the next known section ID.
     */
    const sectionIds =
        getSections()
            .map(s => s.slug)
            .filter(
                id => id !== sectionSlug
            );

    let end = html.length;

    for (const id of sectionIds) {
        const regex = new RegExp(
            `<div\\s+id="${escapeRegExp(id)}"\\s+class="section[^"]*"`,
            'i'
        );

        const match =
            regex.exec(
                html.slice(start + startMatch[0].length)
            );

        if (match) {
            const candidate =
                start +
                startMatch[0].length +
                match.index;

            if (candidate < end) {
                end = candidate;
            }
        }
    }

    /*
     * The principle section is normally the last section,
     * so if this is not it, the above boundary is enough.
     */
    return {
        start,
        end
    };
}


/* =========================================================
   CMS CARD PARSING
   ========================================================= */

function getEntries() {
    const html = readCanon();

    const entries = [];

    /*
     * Only cards carrying data-canon-id are CMS-managed.
     *
     * This is deliberate:
     * existing project canon remains untouched.
     */
    const cardRegex =
        /<div\s+class="card canon-card ([^"]+)"\s+data-status="([^"]+)"\s+data-canon-id="([^"]+)"[^>]*>([\s\S]*?)<\/div>\s*(?=<div\s+class="card canon-card|<\/div>)/gi;

    let match;

    while ((match = cardRegex.exec(html)) !== null) {
        const classes = match[1];
        const status = match[2];
        const id = match[3];
        const body = match[4];

        const titleMatch =
            body.match(
                /<div\s+class="card-title">\s*([\s\S]*?)\s*<\/div>/i
            );

        const textMatch =
            body.match(
                /<div\s+class="card-text">\s*([\s\S]*?)\s*<\/div>/i
            );

        const title =
            titleMatch
                ? stripHtml(titleMatch[1])
                : id;

        const description =
            textMatch
                ? stripHtml(textMatch[1])
                : '';

        const section =
            findEntrySection(
                html,
                match.index
            );

        entries.push({
            id,
            title,
            description,
            status,
            section_slug: section
                ? section.slug
                : '',
            section_name: section
                ? section.name
                : '',
            classes
        });
    }

    return entries;
}

function findEntrySection(html, position) {
    const sections = getSections();

    let best = null;

    for (const section of sections) {
        const range =
            findSectionRangeSafe(
                html,
                section.slug
            );

        if (
            range &&
            position >= range.start &&
            position < range.end
        ) {
            if (
                !best ||
                range.start > best.range.start
            ) {
                best = {
                    section,
                    range
                };
            }
        }
    }

    return best
        ? best.section
        : null;
}

function findSectionRangeSafe(
    html,
    slug
) {
    try {
        return findSectionRange(
            html,
            slug
        );
    } catch {
        return null;
    }
}

function findEntry(query) {
    const q =
        cleanString(query, 300)
            .toLowerCase();

    const entries =
        getEntries();

    return (
        entries.find(
            e => e.title.toLowerCase() === q
        ) ||
        entries.find(
            e => e.id.toLowerCase() === q
        ) ||
        entries.find(
            e => e.title.toLowerCase().includes(q)
        )
    );
}


/* =========================================================
   CARD CREATION
   ========================================================= */

function createCanonCard({
    title,
    description,
    status,
    id
}) {
    const safeTitle =
        escapeHtml(title);

    const safeDescription =
        escapeHtml(description);

    return `
                    <div
                        class="card canon-card ${status}"
                        data-status="${status}"
                        data-canon-id="${id}"
                    >

                        <div class="card-title">
                            ${safeTitle}
                        </div>

                        <div class="card-text">
                            ${safeDescription}
                        </div>

                    </div>
`;
}


/* =========================================================
   ADD ENTRY
   ========================================================= */

function addEntry({
    title,
    description,
    status,
    sectionSlug,
    username
}) {
    if (!validStatus(status)) {
        throw new Error(
            'Invalid canon status.'
        );
    }

    const section =
        findSectionByName(sectionSlug);

    if (!section) {
        throw new Error(
            `Section "${sectionSlug}" not found.`
        );
    }

    const id =
        generateId();

    const result =
        modifyCanon(html => {
            const range =
                findSectionRange(
                    html,
                    section.slug
                );

            const sectionHtml =
                html.slice(
                    range.start,
                    range.end
                );

            /*
             * Prefer inserting into an existing card-grid.
             */
            const cardGridRegex =
                /<div\s+class="card-grid[^"]*"[^>]*>/i;

            const gridMatch =
                sectionHtml.match(
                    cardGridRegex
                );

            const card =
                createCanonCard({
                    title,
                    description,
                    status,
                    id
                });

            let updatedSection;

            if (gridMatch) {
                const insertAt =
                    gridMatch.index +
                    gridMatch[0].length;

                updatedSection =
                    sectionHtml.slice(
                        0,
                        insertAt
                    ) +
                    '\n' +
                    card +
                    sectionHtml.slice(
                        insertAt
                    );

            } else {
                /*
                 * For text-only sections, create a card
                 * grid immediately before section-content
                 * closes.
                 */
                const contentEnd =
                    sectionHtml.lastIndexOf(
                        '</div>'
                    );

                if (contentEnd === -1) {
                    throw new Error(
                        'Could not find a safe insertion point.'
                    );
                }

                const block = `
                    <div class="card-grid">

${card}

                    </div>

`;

                updatedSection =
                    sectionHtml.slice(
                        0,
                        contentEnd
                    ) +
                    block +
                    sectionHtml.slice(
                        contentEnd
                    );
            }

            return (
                html.slice(
                    0,
                    range.start
                ) +
                updatedSection +
                html.slice(
                    range.end
                )
            );
        });

    addRevision({
        action: 'ADD',
        target: 'canon',
        targetId: id,
        details:
            `${status.toUpperCase()}: ${title} → ${section.name}`,
        username
    });

    return {
        id,
        title,
        description,
        status,
        section_slug: section.slug,
        section_name: section.name,
        backup: result.backup
    };
}


/* =========================================================
   UPDATE ENTRY
   ========================================================= */

function updateEntry(
    id,
    patch,
    username
) {
    const entries =
        getEntries();

    const entry =
        entries.find(
            e => e.id === id
        );

    if (!entry) {
        throw new Error(
            'Canon entry not found.'
        );
    }

    if (
        patch.status !== undefined &&
        !validStatus(patch.status)
    ) {
        throw new Error(
            'Invalid canon status.'
        );
    }

    const newSection =
        patch.sectionSlug !== undefined
            ? findSectionByName(
                patch.sectionSlug
            )
            : null;

    if (
        patch.sectionSlug !== undefined &&
        !newSection
    ) {
        throw new Error(
            'Destination section not found.'
        );
    }

    const targetSection =
        newSection || {
            slug: entry.section_slug,
            name: entry.section_name
        };

    const newTitle =
        patch.title !== undefined
            ? patch.title
            : entry.title;

    const newDescription =
        patch.description !== undefined
            ? patch.description
            : entry.description;

    const newStatus =
        patch.status !== undefined
            ? patch.status
            : entry.status;

    /*
     * Find the exact CMS card.
     */
    const cardRegex =
        new RegExp(
            `<div\\s+class="card\\s+canon-card\\s+[^"]*"\\s+data-status="[^"]*"\\s+data-canon-id="${escapeRegExp(entry.id)}"[^>]*>[\\s\\S]*?<\\/div>`,
            'i'
        );

    const match =
        readCanon().match(cardRegex);

    if (!match) {
        throw new Error(
            'Could not locate the canon card in the HTML.'
        );
    }

    const replacement =
        createCanonCard({
            title: newTitle,
            description: newDescription,
            status: newStatus,
            id: entry.id
        });

    const result =
        modifyCanon(html => {
            if (
                newSection &&
                newSection.slug !==
                entry.section_slug
            ) {
                /*
                 * Remove from current location.
                 */
                const removed =
                    html.replace(
                        cardRegex,
                        ''
                    );

                /*
                 * Insert into destination.
                 */
                const range =
                    findSectionRange(
                        removed,
                        newSection.slug
                    );

                const sectionHtml =
                    removed.slice(
                        range.start,
                        range.end
                    );

                const gridRegex =
                    /<div\s+class="card-grid[^"]*"[^>]*>/i;

                const gridMatch =
                    sectionHtml.match(
                        gridRegex
                    );

                if (!gridMatch) {
                    throw new Error(
                        'Destination section has no card grid.'
                    );
                }

                const insertAt =
                    gridMatch.index +
                    gridMatch[0].length;

                const updatedSection =
                    sectionHtml.slice(
                        0,
                        insertAt
                    ) +
                    '\n' +
                    replacement +
                    sectionHtml.slice(
                        insertAt
                    );

                return (
                    removed.slice(
                        0,
                        range.start
                    ) +
                    updatedSection +
                    removed.slice(
                        range.end
                    )
                );
            }

            return html.replace(
                cardRegex,
                replacement
            );
        });

    addRevision({
        action: 'UPDATE',
        target: 'canon',
        targetId: entry.id,
        details:
            `${entry.title} → ${newTitle}`,
        username
    });

    return {
        ...entry,
        title: newTitle,
        description: newDescription,
        status: newStatus,
        section_slug: targetSection.slug,
        section_name: targetSection.name,
        backup: result.backup
    };
}


/* =========================================================
   DELETE ENTRY
   ========================================================= */

function deleteEntry(
    id,
    username
) {
    const entry =
        getEntries().find(
            e => e.id === id
        );

    if (!entry) {
        throw new Error(
            'Canon entry not found.'
        );
    }

    const cardRegex =
        new RegExp(
            `<div\\s+class="card\\s+canon-card\\s+[^"]*"\\s+data-status="[^"]*"\\s+data-canon-id="${escapeRegExp(entry.id)}"[^>]*>[\\s\\S]*?<\\/div>`,
            'i'
        );

    const result =
        modifyCanon(html => {
            if (!cardRegex.test(html)) {
                throw new Error(
                    'Could not locate canon card.'
                );
            }

            return html.replace(
                cardRegex,
                ''
            );
        });

    addRevision({
        action: 'DELETE',
        target: 'canon',
        targetId: id,
        details: entry.title,
        username
    });

    return {
        ok: true,
        title: entry.title,
        backup: result.backup
    };
}


/* =========================================================
   CREATE SECTION
   ========================================================= */

function createSection(
    name,
    username
) {
    const cleanName =
        cleanString(name, 100);

    const slug =
        slugify(cleanName);

    if (!slug) {
        throw new Error(
            'Invalid section name.'
        );
    }

    if (
        getSections().some(
            s => s.slug === slug
        )
    ) {
        throw new Error(
            'Section already exists.'
        );
    }

    modifyCanon(html => {
        const section = `

        <!-- =================================================
             CMS SECTION — ${escapeHtml(cleanName)}
             ================================================= -->

        <div id="${slug}" class="section">

            <div class="section-header">

                <div class="section-number">
                    —
                </div>

                <div class="section-title">
                    ${escapeHtml(cleanName)}
                </div>

            </div>

            <div class="section-content">

                <div class="card-grid">

                </div>

            </div>

        </div>

`;

        const principleIndex =
            html.indexOf(
                '<div id="principle"'
            );

        if (principleIndex === -1) {
            throw new Error(
                'Could not find project principle section.'
            );
        }

        return (
            html.slice(
                0,
                principleIndex
            ) +
            section +
            html.slice(
                principleIndex
            )
        );
    });

    addRevision({
        action: 'SECTION',
        target: 'section',
        targetId: slug,
        details: cleanName,
        username
    });

    return {
        slug,
        name: cleanName
    };
}


/* =========================================================
   AUTH
   ========================================================= */

function requireOwner(
    req,
    res,
    next
) {
    if (
        req.session.user?.role !==
        'owner'
    ) {
        return res
            .status(401)
            .json({
                error:
                    'Owner authentication required'
            });
    }

    next();
}

/* =========================================================
   AUTH API
   ========================================================= */

app.get(
    '/api/auth/me',
    (req, res) => {
        res.json({
            authenticated:
                req.session.user?.role ===
                'owner',
            username:
                req.session.user?.username ||
                null
        });
    }
);

app.post(
    '/api/auth/login',
    (req, res) => {
        try {
            const username =
                cleanString(
                    req.body.username,
                    100
                );

            const password =
                cleanString(
                    req.body.password,
                    500
                );

            const user =
                db.prepare(
                    'SELECT * FROM users WHERE username=?'
                ).get(username);

            if (
                !user ||
                !bcrypt.compareSync(
                    password,
                    user.password_hash
                )
            ) {
                return res
                    .status(401)
                    .json({
                        error:
                            'Invalid credentials'
                    });
            }

            req.session.regenerate(
                err => {
                    if (err) {
                        return res
                            .status(500)
                            .json({
                                error:
                                    'Could not start session'
                            });
                    }

                    req.session.user = {
                        id: user.id,
                        username: user.username,
                        role: user.role
                    };

                    res.json({
                        ok: true
                    });
                }
            );

        } catch (error) {
            res
                .status(400)
                .json({
                    error:
                        error.message
                });
        }
    }
);

app.post(
    '/api/auth/logout',
    (req, res) => {
        req.session.destroy(
            () => res.json({
                ok: true
            })
        );
    }
);

/* =========================================================
   COMMAND ENGINE
   ========================================================= */

app.post(
    '/api/admin/command',
    requireOwner,
    (req, res) => {
        try {
            const input =
                cleanString(
                    req.body.command,
                    1000
                );

            const username =
                req.session.user.username;

            /*
             * ADD STATUS "Title" : "Description"
             * TO "Section"
             */
            let match =
                input.match(
                    /^ADD\s+(IDEA|PROVISIONAL|LOCKED|REJECTED)\s+"([^"]+)"\s*:\s*"([^"]+)"(?:\s+TO\s+"([^"]+)")?$/i
                );

            if (match) {
                const status =
                    match[1].toLowerCase();

                const title =
                    match[2];

                const description =
                    match[3];

                const section =
                    match[4]
                        ? findSectionByName(
                            match[4]
                        )
                        : findSectionByName(
                            'canon'
                        );

                if (!section) {
                    throw new Error(
                        'Section not found.'
                    );
                }

                const entry =
                    addEntry({
                        title,
                        description,
                        status,
                        sectionSlug:
                            section.slug,
                        username
                    });

                return res.json({
                    ok: true,
                    message:
                        `Added ${entry.title}`,
                    entry
                });
            }


            /*
             * ADD STATUS "Title"
             */
            match =
                input.match(
                    /^ADD\s+(IDEA|PROVISIONAL|LOCKED|REJECTED)\s+"([^"]+)"(?:\s+TO\s+"([^"]+)")?$/i
                );

            if (match) {
                const status =
                    match[1].toLowerCase();

                const title =
                    match[2];

                const section =
                    match[3]
                        ? findSectionByName(
                            match[3]
                        )
                        : findSectionByName(
                            'canon'
                        );

                if (!section) {
                    throw new Error(
                        'Section not found.'
                    );
                }

                const entry =
                    addEntry({
                        title,
                        description: title,
                        status,
                        sectionSlug:
                            section.slug,
                        username
                    });

                return res.json({
                    ok: true,
                    message:
                        `Added ${entry.title}`,
                    entry
                });
            }


            /*
             * SECTION "Name"
             */
            match =
                input.match(
                    /^SECTION\s+"([^"]+)"$/i
                );

            if (match) {
                const section =
                    createSection(
                        match[1],
                        username
                    );

                return res.json({
                    ok: true,
                    message:
                        `Created section ${section.name}`,
                    section
                });
            }


            /*
             * MOVE "Title" TO "Section"
             */
            match =
                input.match(
                    /^MOVE\s+"([^"]+)"\s+TO\s+"([^"]+)"$/i
                );

            if (match) {
                const entry =
                    findEntry(match[1]);

                const section =
                    findSectionByName(
                        match[2]
                    );

                if (!entry) {
                    throw new Error(
                        'Entry not found.'
                    );
                }

                if (!section) {
                    throw new Error(
                        'Section not found.'
                    );
                }

                const updated =
                    updateEntry(
                        entry.id,
                        {
                            sectionSlug:
                                section.slug
                        },
                        username
                    );

                return res.json({
                    ok: true,
                    message:
                        `Moved ${entry.title} to ${section.name}`,
                    entry: updated
                });
            }


            /*
             * STATUS "Title" LOCKED
             */
            match =
                input.match(
                    /^STATUS\s+"([^"]+)"\s+(IDEA|PROVISIONAL|LOCKED|REJECTED)$/i
                );

            if (match) {
                const entry =
                    findEntry(match[1]);

                if (!entry) {
                    throw new Error(
                        'Entry not found.'
                    );
                }

                const updated =
                    updateEntry(
                        entry.id,
                        {
                            status:
                                match[2].toLowerCase()
                        },
                        username
                    );

                return res.json({
                    ok: true,
                    message:
                        `Changed ${entry.title} to ${updated.status}`,
                    entry: updated
                });
            }


            /*
             * DELETE "Title"
             */
            match =
                input.match(
                    /^DELETE\s+"([^"]+)"$/i
                );

            if (match) {
                const entry =
                    findEntry(match[1]);

                if (!entry) {
                    throw new Error(
                        'Entry not found.'
                    );
                }

                const result =
                    deleteEntry(
                        entry.id,
                        username
                    );

                return res.json({
                    ok: true,
                    message:
                        `Deleted ${entry.title}`,
                    result
                });
            }


            /*
             * LIST
             */
            if (
                /^LIST(?:\s+(.*))?$/i.test(
                    input
                )
            ) {
                const listMatch =
                    input.match(
                        /^LIST(?:\s+(.*))?$/i
                    );

                let rows =
                    getEntries();

                if (listMatch[1]) {
                    const query =
                        listMatch[1]
                            .trim();

                    const section =
                        findSectionByName(
                            query
                        );

                    if (section) {
                        rows =
                            rows.filter(
                                e =>
                                    e.section_slug ===
                                    section.slug
                            );
                    } else if (
                        validStatus(
                            query.toLowerCase()
                        )
                    ) {
                        rows =
                            rows.filter(
                                e =>
                                    e.status ===
                                    query.toLowerCase()
                            );
                    }
                }

                return res.json({
                    ok: true,
                    message:
                        rows.map(
                            e =>
                                `${e.title} [${e.status}] — ${e.section_name}`
                        ).join('\n') ||
                        'No entries found.',
                    entries: rows
                });
            }


            /*
             * HELP
             */
            if (
                /^HELP$/i.test(
                    input
                )
            ) {
                return res.json({
                    ok: true,
                    message:
                        'ADD IDEA "Title" : "Description" TO "Section" | SECTION "Name" | MOVE "Title" TO "Section" | STATUS "Title" LOCKED | DELETE "Title" | LIST [Section/Status] | HELP'
                });
            }


            throw new Error(
                'Unknown command. Type HELP.'
            );

        } catch (error) {
            res
                .status(400)
                .json({
                    error:
                        error.message
                });
        }
    }
);
